-- Main para testar Example2.txt para ler input

main :: IO ()
main = do
    source <- readFile "TesteAtribuicao.txt"  --decide qual ficheiro de teste usar
    let tokens = alexScanTokens source

    case parse tokens of
        PascalProgram _ _ statements -> do
            _ <- runStatements [] statements
            pure ()

runStatements :: Env -> [Stmt] -> IO Env
runStatements env [] = pure env
runStatements env (stmt:rest) = do
    newEnv <- runStatement env stmt
    runStatements newEnv rest

runStatement :: Env -> Stmt -> IO Env
runStatement env (WriteStmt args) = do
    mapM_ (putStr . renderWriteArg env) args
    putStrLn ""
    pure env
runStatement env (ReadStmt names) =
    readVariables env names
runStatement env ReadLineStmt = do
    _ <- getLine
    pure env
runStatement env (AssignStmt name expr) =
    pure (setVariable name (evalExpr env expr) env)

readVariables :: Env -> [String] -> IO Env
readVariables env [] = pure env
readVariables env (name:names) = do
    value <- getLine
    readVariables (setVariable name value env) names

renderWriteArg :: Env -> WriteArg -> String
renderWriteArg env (WriteExpr expr) =
    evalExpr env expr
renderWriteArg _ (WriteFormat _ _ _) =
    error "A formatação de writeln não está implementada neste teste."

evalExpr :: Env -> Expr -> String
evalExpr _ (StringExpr value) =
    value
evalExpr env (VariableExpr name) =
    case lookup name env of
        Just value -> value
        Nothing -> error ("Variável sem valor: " ++ name)
evalExpr _ (NumberExpr number) =
    show number
evalExpr _ (BooleanExpr value) =
    show value
evalExpr env (BinaryExpr op left right) =
    let l = evalNumber env left
        r = evalNumber env right
    in case op of
        Add             -> show (l + r)
        Subtract        -> show (l - r)
        Multiply        -> show (l * r)
        Divide
            | r == 0    -> error "Divisão por zero."
            | otherwise -> show (l / r)
        EqualTo         -> show (l == r)
        NotEqualTo      -> show (l /= r)
        LessThan        -> show (l < r)
        LessEqualTo     -> show (l <= r)
        GreaterThan     -> show (l > r)
        GreaterEqualTo  -> show (l >= r)

evalNumber :: Env -> Expr -> Double
evalNumber _ (NumberExpr number) =
    number
evalNumber env (VariableExpr name) =
    case lookup name env of
        Just value ->
            case reads value of
                [(number, "")] -> number
                _ -> error ("A variável " ++ name ++ " não contém um número.")
        Nothing ->
            error ("Variável sem valor: " ++ name)
evalNumber env (BinaryExpr op left right) =
    let l = evalNumber env left
        r = evalNumber env right
    in case op of
        Add             -> l + r
        Subtract        -> l - r
        Multiply        -> l * r
        Divide
            | r == 0    -> error "Divisão por zero."
            | otherwise -> l / r
        _ -> error "Operador de comparação usado numa expressão numérica."
evalNumber _ _ =
    error "A expressão não é numérica."

setVariable :: String -> String -> Env -> Env
setVariable name value [] =
    [(name, value)]
setVariable name value ((key, oldValue):rest)
    | name == key = (key, value) : rest
    | otherwise = (key, oldValue) : setVariable name value rest
